    (function () {
      var typedEmail = <?!= JSON.stringify(typedEmail) ?>;
      var returnOrigin = <?!= JSON.stringify(returnOrigin) ?>;
      var statusEl = document.getElementById("status");
      var detailEl = document.getElementById("detail");

      function postToOpener(payload) {
        var msg = Object.assign({ type: "wushuang-auth-bridge" }, payload);
        try {
          if (window.opener && !window.opener.closed) {
            window.opener.postMessage(msg, returnOrigin || "*");
          }
        } catch (e) {}
        try {
          if (window.parent && window.parent !== window) {
            window.parent.postMessage(msg, returnOrigin || "*");
          }
        } catch (e2) {}
      }

      function finish(result) {
        if (result && result.ok && result.match) {
          statusEl.className = "ok";
          statusEl.textContent = "核對成功。可以關閉本視窗。";
          detailEl.textContent = "Google：" + (result.googleEmail || "");
        } else {
          statusEl.className = "err";
          statusEl.textContent = (result && result.error) || "核對失敗";
          var parts = [];
          if (result && result.googleEmail) {
            parts.push(
              "Google：" + result.googleEmail + " ≠ 輸入：" + (result.typedEmail || typedEmail)
            );
          }
          parts.push("若頁面曾空白：請用學校帳單獨開本連結完成授權後再試。");
          detailEl.textContent = parts.join(" ");
        }
        postToOpener(result || { ok: false, error: "核對失敗", match: false });
        setTimeout(function () {
          try { window.close(); } catch (e) {}
        }, 1600);
      }

      if (!typedEmail) {
        finish({ ok: false, error: "缺少輸入 email。", match: false, typedEmail: "" });
        return;
      }

      if (typeof google === "undefined" || !google.script || !google.script.run) {
        finish({
          ok: false,
          error: "未能載入 Google Apps Script。請用學校帳單獨開此頁完成授權。",
          match: false,
          typedEmail: typedEmail,
        });
        return;
      }

      google.script.run
        .withSuccessHandler(finish)
        .withFailureHandler(function (err) {
          finish({
            ok: false,
            error:
              "核對失敗：" +
              String(err && err.message ? err.message : err) +
              "。請用學校帳單獨開此頁完成授權後再試。",
            match: false,
            typedEmail: typedEmail,
          });
        })
        .verifyAndLog(typedEmail);
    })();
  </script>
